    function addPeserta(fokus) {
      if (kartuPeserta().length >= MAX_PESERTA) {

        return null;

    function handleFile(input) {
      var file = input.files && input.files[0];
      resetPreview();
      if (!file) return;

      // Validasi tipe file
      if (ALLOWED_TYPES.indexOf(file.type) === -1) {
        showAlert('danger', 'Format file tidak didukung (' + (file.type || 'tidak dikenal') +
          '). Gunakan foto berformat JPG atau PNG.');
        input.value = '';
        return;
      }

      // Validasi ukuran file
      if (file.size > MAX_SIZE) {
        showAlert('danger', 'Ukuran file ' + bytesToSize(file.size) +
          ' melebihi batas 5 MB. Silakan kompres atau kecilkan foto terlebih dahulu.');
        input.value = '';
        return;
      }

      // Konversi ke base64 untuk dikirim ke server + tampilkan preview
      var reader = new FileReader();
      reader.onload = function (ev) {
        var dataUrl = ev.target.result;
        selectedFile = {
          name: file.name,
          mime: file.type,
          size: file.size,
          base64: dataUrl.substring(dataUrl.indexOf(',') + 1) // buang prefix "data:...;base64,"
        };
        el('previewImg').src = dataUrl;
        el('previewName').textContent = file.name;
        el('previewSize').textContent = bytesToSize(file.size) + ' \u2014 siap diunggah';
        el('filePreview').style.display = 'block';
      };
      reader.onerror = function () {
        showAlert('danger', 'Gagal membaca file. Silakan coba pilih file lagi.');
        input.value = '';
      };
      reader.readAsDataURL(file);
    }

    function resetPreview() {
      selectedFile = null;
      el('filePreview').style.display = 'none';
      el('previewImg').src = '';
    }

    /* =========================================================

     * sehingga pendamping bisa mulai mengisi dari awal. Hanya
     * membersihkan HALAMAN ini - data yang sudah terkirim ke

    function resetFormulir() {
      if (isSubmitting) return; // jangan reset saat data sedang dikirim

      // Deteksi apakah ada isian yang akan hilang (untuk konfirmasi)
      var adaIsian =
        ['namaSekolah', 'namaPendamping', 'nomorWa', 'jumlahBayar', 'email']
          .some(function (id) { return el(id).value.trim() !== ''; }) ||
        kartuPeserta().length > 1 ||
        Array.prototype.some.call(kartuPeserta(), function (card) {
          return card.querySelector('.p-nama').value.trim() !== '' ||
                 card.querySelector('.p-kelas').value.trim() !== '' ||
                 card.querySelector('.p-mata').value !== '';
        }) ||
        selectedFile !== null;

      if (adaIsian &&
          !window.confirm('Kosongkan semua isian pada formulir ini?\n\n' +
            'Semua data yang sudah diketik (termasuk daftar peserta dan bukti bayar) ' +
            'akan dihapus dari halaman ini.\n\n' +
            'Catatan: pendaftaran yang SUDAH TERKIRIM ke panitia tidak terpengaruh.')) {
        return;
      }

      // 1) Kosongkan kolom statis + status validasi Bootstrap
      var form = el('formPendaftaran');
      form.reset();
      form.classList.remove('was-validated');

      // 2) Kosongkan daftar peserta: hapus semua kartu, buat 1 kartu baru
      el('daftarPeserta').innerHTML = '';
      addPeserta(false);

      // 3) Kosongkan pilihan file bukti bayar + preview
      el('buktiBayar').value = '';
      resetPreview();

      // 4) Kosongkan preview Rupiah & kotak alert
      el('bayarPreview').textContent = '\u00A0';

      window.scrollTo({ top: 0, behavior: 'smooth' });

    function handleFileTtd(input) {

      resetPreviewTtd();

          '). Gunakan foto berformat JPG/PNG atau file PDF.');
        input.value = '';
        return;
      }

      // Validasi ukuran file
      if (file.size > MAX_SIZE) {
        tampilkanStatusTtd('danger', 'Ukuran file ' + bytesToSize(file.size) +

        return;

    function kirimFormulirTtd() {
      if (!nomorPendaftaranSaatIni) {
        tampilkanStatusTtd('warning', 'Nomor pendaftaran belum tersedia. Buka halaman konfirmasi pendaftaran Anda terlebih dahulu.');
        return;

      if (!selectedFileTtd) {
        tampilkanStatusTtd('warning', 'Pilih dulu foto/scan formulir yang sudah ditandatangani (klik area unggah di atas).');

      }

      var btn = el('btnKirimTtd');
      btn.disabled = true;
      btn.innerHTML = '<span class="spinner-border spinner-border-sm me-2"></span>Mengunggah...';

      google.script.run
        .withSuccessHandler(function (res) {
          btn.disabled = false;
          btn.innerHTML = '<i class="bi bi-upload me-1"></i> Unggah Ulang Formulir';
          if (!res || !res.success) {
            tampilkanStatusTtd('danger', (res && res.message) ? res.message : 'Gagal mengunggah formulir. Silakan coba lagi.');
            return;
          }
          tampilkanStatusTtd('success',

            '<a href="' + res.fileUrl + '" target="_blank">Lihat file di Google Drive</a>. ' +

          el('buktiFormulirTtd').value = '';

          btn.disabled = false;
          btn.innerHTML = '<i class="bi bi-upload me-1"></i> Unggah Formulir Tanda Tangan';
          tampilkanStatusTtd('danger', 'Gagal terhubung ke server. Periksa koneksi internet lalu coba lagi. (' +
            (err && err.message ? err.message : err) + ')');
        })
        .uploadFormulirTtd({
          nomorPendaftaran: nomorPendaftaranSaatIni,
          fileName: selectedFileTtd.name,
          fileMime: selectedFileTtd.mime,
          fileBase64: selectedFileTtd.base64

      if (isSubmitting) return;

      var form = el('formPendaftaran');

      // Validasi HTML5 (Bootstrap was-validated menampilkan pesan merah)
      // - mencakup semua kartu peserta: nama, kelas, dan mata lomba wajib terisi
      if (!form.checkValidity()) {
        form.classList.add('was-validated');
        showAlert('warning', 'Mohon lengkapi semua kolom wajib dengan benar, termasuk data setiap peserta.');
        return;
      }

      // Validasi file terpisah (input file disembunyikan, divalidasi manual)
      if (!selectedFile) {
        showAlert('danger', 'Bukti bayar wajib diunggah (JPG/PNG, maksimal 5 MB).');
        el('dropzone').scrollIntoView({ behavior: 'smooth', block: 'center' });
        return;
      }

      // Kumpulkan daftar peserta dari semua kartu
      var peserta = [];
      kartuPeserta().forEach(function (card) {
        peserta.push({
          nama: card.querySelector('.p-nama').value.trim(),
          kelas: card.querySelector('.p-kelas').value.trim(),
          mataLomba: card.querySelector('.p-mata').value
        });
      });
      if (peserta.length === 0) {
        showAlert('warning', 'Minimal satu peserta harus ditambahkan.');
        return;
      }

      // Susun payload untuk server
      // CATATAN: nomor pendaftaran TIDAK dikirim - diberikan otomatis oleh server
      var payload = {
        namaSekolah: el('namaSekolah').value.trim(),
        namaPendamping: el('namaPendamping').value.trim(),
        nomorWa: el('nomorWa').value.trim(),
        peserta: peserta, // array {nama, kelas, mataLomba}
        jumlahBayar: el('jumlahBayar').value,
        email: el('email').value.trim(),
        fileName: selectedFile.name,
        fileMime: selectedFile.mime,
        fileBase64: selectedFile.base64
      };

      isSubmitting = true;
      setLoading(true);

      google.script.run
        .withSuccessHandler(onSubmitSuccess)
        .withFailureHandler(onSubmitError)
        .submitRegistration(payload);
    }

    function onSubmitSuccess(res) {
      setLoading(false);
      isSubmitting = false;

      if (!res || !res.success) {
        var msg = (res && res.message) ? res.message : 'Terjadi kesalahan saat menyimpan pendaftaran.';
        showAlert('danger', msg);
        return;
      }
      tampilkanSukses(res.data || {});
    }

    function onSubmitError(err) {
      setLoading(false);
      isSubmitting = false;
      showAlert('danger', 'Gagal terhubung ke server. Periksa koneksi internet lalu coba lagi. (' +
        (err && err.message ? err.message : err) + ')');
    }

    /* =========================================================
     * UI: LOADING, ALERT, LAYAR SUKSES + FORMULIR RESMI
     * ========================================================= */
    function setLoading(loading) {
      var btn = el('btnSubmit');
      btn.disabled = loading;
      btn.innerHTML = loading
        ? '<span class="spinner-border spinner-border-sm me-2"></span>Mengirim...'
        : '<i class="bi bi-send me-1"></i> Kirim Pendaftaran';

      // Kunci semua input (termasuk select mata lomba & tombol tambah/hapus) saat proses berjalan
      var inputs = document.querySelectorAll(
        '#formPendaftaran input, #formPendaftaran select, #formPendaftaran button');
      Array.prototype.forEach.call(inputs, function (inp) {
        inp.disabled = loading;
      });
    }

    function showAlert(type, message) {
      var wrap = el('alertBox');
      wrap.innerHTML =
        '<div class="alert alert-' + type + ' alert-dismissible fade show" role="alert">' +
          message +
          '<button type="button" class="btn-close" data-bs-dismiss="alert" aria-label="Tutup"></button>' +
        '</div>';
      wrap.scrollIntoView({ behavior: 'smooth', block: 'center' });
    }

    function escHtml(s) {
      return String(s == null ? '' : s)
        .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
    }

    function tampilkanSukses(data) {
      var tglCetak = new Date().toLocaleDateString('id-ID', { day: 'numeric', month: 'long', year: 'numeric' });

      // --- Isi banner sukses ---
      el('suksesNomor').textContent = data.nomorPendaftaran || '-';

      // Simpan nomor pendaftaran untuk fitur upload formulir tanda tangan
      nomorPendaftaranSaatIni = data.nomorPendaftaran || '';
      resetUploadTtd();

      if (data.emailSent === false) {
        el('suksesCatatanEmail').innerHTML =
          '<i class="bi bi-exclamation-triangle me-1"></i>Catatan: email konfirmasi gagal terkirim, ' +
          'namun data Anda sudah tersimpan dan akan diproses panitia.';
        el('suksesCatatanEmail').classList.add('text-danger');
      }

      // --- Isi formulir resmi (replika formulir PENTAS PAI SD) ---
      el('cetakNomor').textContent = data.nomorPendaftaran || '-';
      el('cetakTanggal').textContent = 'Dicetak: ' + tglCetak;
      el('cetakSekolah').textContent = data.namaSekolah || '-';
      el('cetakPendamping').textContent = data.namaPendamping || '-';
      el('cetakWa').textContent = data.nomorWa || '-';

      // Kelompokkan peserta per mata lomba -> isi tabel 12 mata lomba
      var byKat = {};
      (data.peserta || []).forEach(function (p) {
        if (!byKat[p.mataLomba]) byKat[p.mataLomba] = { nama: [], kelas: [] };
        byKat[p.mataLomba].nama.push(p.nama);
        byKat[p.mataLomba].kelas.push(p.kelas);
      });
      document.querySelectorAll('#tabelPesertaCetak tbody tr').forEach(function (tr, i) {
        tr.classList.remove('terisi');
        tr.querySelector('.no').textContent = i + 1;
        tr.querySelector('.nama').innerHTML = '';
        tr.querySelector('.kelas').innerHTML = '';
        var isi = byKat[tr.getAttribute('data-kat')];
        if (isi) {
          tr.classList.add('terisi');
          tr.querySelector('.nama').innerHTML = isi.nama.map(escHtml).join('<br>');
          tr.querySelector('.kelas').innerHTML = isi.kelas.map(escHtml).join('<br>');
        }
      });

      // Tampilkan layar sukses
      el('alertBox').innerHTML = '';
      el('cardForm').classList.add('d-none');
      el('cardSukses').classList.remove('d-none');
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    /* =========================================================
     * CETAK FORMULIR RESMI (ANTI KOSONG)
     * Google Apps Script menjalankan aplikasi di dalam IFRAME.
     * window.print() dari dalam iframe sering mencetak halaman
     * pembungkusnya yang KOSONG (terutama Chrome) - inilah penyebab
     * "formulir tercetak kosong". Solusi: salin formulir beserta
     * kondisi datanya ke JENDELA BARU yang bersih, lalu cetak dari
     * jendela tersebut. Bila popup diblokir, kembali ke cara lama.
     * ========================================================= */
    function cetakFormulir() {
      var src = el('formulirCetak');
      if (!src) return;

      // Salin kondisi formulir saat ini (identitas & tabel peserta terisi)
      var isiFormulir = src.outerHTML;

      var html =
        '<!DOCTYPE html><html lang="id"><head><meta charset="utf-8">' +
        '<title>Formulir Pendaftaran - ' + escHtml(el('cetakNomor').textContent || '') + '</title>' +
        '<style>' + cssFormulirCetak() + '</style>' +
        '</head><body>' +
        '<div class="toolbar-cetak">' +
          '<span>Pratinjau cetak - jika dialog tidak muncul otomatis, tekan <b>Ctrl+P</b> (Mac: <b>Cmd+P</b>).</span>' +
          '<span>' +
            '<button type="button" onclick="window.print()">Cetak / Simpan PDF</button>' +
            '<button type="button" onclick="window.close()">Tutup</button>' +
          '</span>' +
        '</div>' +
        isiFormulir +
        '</body></html>';

      var w = window.open('', '_blank');
      if (!w) {
        // Popup diblokir browser -> pakai cara lama (cetak di halaman ini).
        // CSS @media print sudah menyembunyikan semua kecuali formulir.
        alert('Popup diblokir browser. Izinkan popup untuk hasil cetak terbaik.\n' +
              'Cetak dilanjutkan di halaman ini sebagai alternatif.');
        window.print();
        return;
      }

      w.document.open();
      w.document.write(html);
      w.document.close();
      w.focus();
      setTimeout(function () {
        try { w.print(); } catch (e) { /* user bisa pakai tombol di toolbar */ }
      }, 400);
    }

    function formatRupiahClient(angka) {
      var n = isNaN(Number(angka)) ? 0 : Number(angka);
      return 'Rp ' + n.toFixed(0).replace(/\B(?=(\d{3})+(?!\d))/g, '.');
    }

    function bytesToSize(bytes) {
      if (bytes < 1024) return bytes + ' B';
      if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' KB';
      return (bytes / 1048576).toFixed(2) + ' MB';
    }
